<script>
/* ---------- 마름모 1분 모션 (2026-11-20 R3 — 열 가지 한 세트) ----------
   대문의 방치 유도(assets/modes.js)는 매 분 00초에 5.0초 동안 심전도·레버를 깨운다. 그 순간 #ecgRow 또는 #slot 에
   data-st="hint" 가 붙고, 다 끝나면 지워진다. 마름모는 그 속성 변화를 그대로 받아 **같은 프레임에** 시작한다 —
   따로 시계를 세지 않으므로 심전도·롤링과 어긋날 수가 없다(장전·오버레이·감속·가려진 화면에서는 modes.js가
   방치 유도를 아예 걸지 않으므로 마름모도 함께 쉰다).
   패턴 열 가지를 분 단위로 하나씩 돌려 쓴다(분 % 10). 모두 4.6초 안에 끝나고, 끝난 자리는 시작한 자리와 똑같다
   (마름모는 90° 대칭이라 180°·360° 회전은 실루엣이 그대로다) — 어느 순간 멈춰도 잔상이 남지 않는다.
   모션이 없을 때는 쉬는 동안 그 자리에 가만히 있다(애니메이션 객체도 남기지 않는다).
   ① 안·밖 반대 회전 ② 글자 위 비행 소멸 ③ 호흡 교차 ④ 엇박 반바퀴
   ⑤ 위아래에서 나타나 결합 ⑥ 획을 차례로 그리기 ⑦ 물결 ⑧ 미세 진동 ⑨ 통통 튀어 착지 ⑩ 표창 회전 */
(function(){
  var lock=document.querySelector('.cctt-lockup'),mark=document.querySelector('.cctt-mark'),
      og=document.querySelector('.cctt-dg-o'),ig=document.querySelector('.cctt-dg-i');
  if(!lock||!mark||!og||!ig||!mark.animate)return;
  var op=[].slice.call(og.querySelectorAll('path')),ip=[].slice.call(ig.querySelectorAll('path')),
      PATH=op.concat(ip);
  var mq=window.matchMedia?window.matchMedia('(prefers-reduced-motion: reduce)'):null;
  function reduce(){return !!(mq&&mq.matches);}
  /* 우아하고 절제된 이징 — 처음에 살짝 늦게 떠나 끝에서 길게 멎는다(심전도 방치 유도와 같은 결). */
  var EASE='cubic-bezier(.42,0,.26,1)',EASE2='cubic-bezier(.5,0,.2,1)';
  var NAMES=['반대 회전','비행 소멸','호흡 교차','엇박 반바퀴','위아래 결합','획 그리기','물결','미세 진동','통통 착지','표창 회전'];
  var runs=[],busy=false,idx=-1,startedAt=0,minute=-1,endT=0,reset=null;
  /* ---- 1~4 : R2에서 확정된 네 가지(그대로) ---- */
  /* 1) 안·밖이 서로 반대 방향으로 한 바퀴 — 바깥은 시계, 안쪽은 반시계(살짝 늦게 따라 돈다) */
  function p0(){return[
    [og,[{transform:'rotate(0deg)'},{transform:'rotate(360deg)'}],{duration:2300,easing:EASE}],
    [ig,[{transform:'rotate(0deg)'},{transform:'rotate(-360deg)'}],{duration:2300,delay:140,easing:EASE}]];}
  /* 2) 글자 위를 한 번 비행해 오른쪽 끝에서 사라지고, 제자리에서 부드럽게 다시 떠오른다 */
  function p1(){
    var W=lock.clientWidth||lock.getBoundingClientRect().width,M=mark.getBoundingClientRect().width||0;
    var D=Math.max(24,W-M+4).toFixed(1);
    return[[mark,[
      {transform:'translateX(0px)',opacity:1,offset:0,easing:'cubic-bezier(.4,0,.6,1)'},
      {transform:'translateX('+(D*.72)+'px)',opacity:.5,offset:.34,easing:'cubic-bezier(.45,0,.7,1)'},
      {transform:'translateX('+D+'px)',opacity:0,offset:.5,easing:'linear'},
      {transform:'translateX(0px)',opacity:0,offset:.58,easing:'linear'},
      {transform:'translateX(0px)',opacity:1,offset:1}],{duration:3200,easing:'linear'}]];}
  /* 3) 숨을 들이쉬듯 아주 조금 커지며 90° 교차, 잠깐 머물다 제자리로 — 소리가 나지 않는 회전 */
  function p2(){return[
    [og,[{transform:'rotate(0deg) scale(1)',offset:0,easing:EASE2},{transform:'rotate(90deg) scale(1.055)',offset:.36,easing:'linear'},
         {transform:'rotate(90deg) scale(1.055)',offset:.62,easing:EASE2},{transform:'rotate(0deg) scale(1)',offset:1}],{duration:3300}],
    [ig,[{transform:'rotate(0deg) scale(1)',offset:0,easing:EASE2},{transform:'rotate(-90deg) scale(1.075)',offset:.4,easing:'linear'},
         {transform:'rotate(-90deg) scale(1.075)',offset:.62,easing:EASE2},{transform:'rotate(0deg) scale(1)',offset:1}],{duration:3300}]];}
  /* 4) 두 겹이 차례로 반 바퀴씩 엇박자로 돌고 함께 제자리에 선다 */
  function p3(){return[
    [og,[{transform:'rotate(0deg)',offset:0,easing:EASE2},{transform:'rotate(180deg)',offset:.44,easing:'linear'},{transform:'rotate(180deg)',offset:1}],{duration:2800}],
    [ig,[{transform:'rotate(0deg)',offset:0,easing:'linear'},{transform:'rotate(0deg)',offset:.16,easing:EASE2},{transform:'rotate(-180deg)',offset:.66,easing:'linear'},{transform:'rotate(-180deg)',offset:1}],{duration:2800}]];}
  /* ---- 5~10 : 2026-11-20 R3에서 더한 여섯 가지 ---- */
  /* 5) 자취를 지운 뒤 바깥은 **위에서**, 안쪽은 **아래에서** 나타나 가운데에서 맞붙는다(닿을 때 아주 조금 늘었다 줄어든다).
     위·아래로 뛰므로 그동안만 .cctt-leap으로 로고 상자의 문을 열어 둔다 — 잘린 자리가 보이지 않게. */
  function p4(){
    var D=30;
    function half(dy){
      return [
        {transform:'translateY(0px) scale(1)',opacity:1,offset:0,easing:'cubic-bezier(.4,0,.65,1)'},
        {transform:'translateY(0px) scale(1)',opacity:0,offset:.13,easing:'linear'},
        {transform:'translateY('+dy+'px) scale(1.06)',opacity:0,offset:.24,easing:'linear'},
        {transform:'translateY('+dy+'px) scale(1.06)',opacity:1,offset:.33,easing:'linear'},
        {transform:'translateY(0px) scale(1)',opacity:1,offset:.72,easing:'cubic-bezier(.24,1.02,.36,1)'},
        {transform:'translateY(0px) scale(1)',opacity:1,offset:1}];
    }
    return {defs:[[og,half(-D),{duration:2900}],[ig,half(D),{duration:2900,delay:70}]],leap:true};
  }
  /* 6) 변(획)대로 차례로 그어진다 — 겹 네 개를 0.68초 간격으로 dashoffset을 0까지 밀어 그린다.
     끝나면 인라인 dash 값을 지워 정지 상태(원래의 온전한 선)와 완전히 같게 만든다. */
  function dashSet(el){
    var L=640;try{L=Math.max(1,el.getTotalLength());}catch(e){}
    el.style.strokeDasharray=L+'px';el.style.strokeDashoffset=L+'px';
    return L.toFixed(1);
  }
  function dashClear(){for(var i=0;i<PATH.length;i++){PATH[i].style.strokeDasharray='';PATH[i].style.strokeDashoffset='';}}
  function p5(){
    var els=[op[0],op[1],ip[0],ip[1]],defs=[],i,L;
    for(i=0;i<els.length;i++){
      L=dashSet(els[i]);
      defs.push([els[i],[{strokeDashoffset:L+'px'},{strokeDashoffset:'0px'}],
        {duration:1050,delay:i*680,easing:'cubic-bezier(.45,0,.25,1)',fill:'both'}]);
    }
    return {defs:defs,reset:dashClear};
  }
  /* 7) 물결 — 두 겹이 물결을 엇갈려 타고 지나간다. 양 끝에서 진폭을 0으로 조여(envelope) 시작·끝 자리가 정확히 같다. */
  function p6(){
    var N=28;
    function ripple(phase){
      var f=[],j,p,ph,env;
      for(j=0;j<=N;j++){
        p=j/N;env=Math.sin(Math.PI*p);ph=p*Math.PI*4+phase;
        f.push({transform:'translateY('+(Math.sin(ph)*1.8*env).toFixed(2)+'px) rotate('+(Math.sin(ph)*1.5*env).toFixed(2)+'deg) scale('+(1+Math.sin(ph+Math.PI/2)*.018*env).toFixed(4)+')',offset:p});
      }
      return f;
    }
    return {defs:[[og,ripple(0),{duration:3100,easing:'linear'}],[ig,ripple(1.9),{duration:3100,easing:'linear',delay:60}]]};
  }
  /* 8) 미세 진동 — 손끝으로 톡 건드린 것처럼 여덟 번 떨고 곧바로 정위치로 선다(끝에서 진폭이 0으로 수렴). */
  function p7(){
    function shake(dir,delay){
      var f=[],j,p,a,dec,env;
      for(j=0;j<=16;j++){
        p=j/16;dec=Math.pow(1-p,1.5);env=Math.min(1,p*20);
        a=Math.sin(p*Math.PI*8)*1.3*dec*env*dir;
        f.push({transform:'translateX('+a.toFixed(2)+'px) rotate('+(a*.35).toFixed(2)+'deg)',offset:p});
      }
      return [f,{duration:1000,easing:'linear',delay:delay}];
    }
    var A=shake(1,0),B=shake(-1,55);
    return {defs:[[og,A[0],A[1]],[ig,B[0],B[1]]]};
  }
  /* 9) 픽사 로고 흉내 — 위에서 두 번 통통 튀며 내려와 착지한다(떨어질 때 살짝 늘고, 닿을 때 눌렸다 풀린다). */
  function p8(){
    function drop(dy,delay){
      return [[
        {transform:'translateY('+dy+'px) scale(.94,.90)',opacity:0,offset:0,easing:'cubic-bezier(.4,0,.9,1)'},
        {transform:'translateY('+dy+'px) scale(.96,.94)',opacity:1,offset:.09,easing:'cubic-bezier(.45,0,.95,1)'},
        {transform:'translateY(0px) scale(1,1)',offset:.31,easing:'cubic-bezier(.2,.9,.3,1)'},
        {transform:'translateY(0px) scale(1.12,.86)',offset:.37,easing:'cubic-bezier(.3,0,.4,1)'},
        {transform:'translateY(-15px) scale(.96,1.06)',offset:.50,easing:'cubic-bezier(.25,.8,.4,1)'},
        {transform:'translateY(0px) scale(1,1)',offset:.64,easing:'cubic-bezier(.2,.9,.3,1)'},
        {transform:'translateY(0px) scale(1.05,.95)',offset:.69,easing:'cubic-bezier(.3,0,.4,1)'},
        {transform:'translateY(-6px) scale(.99,1.01)',offset:.78,easing:'cubic-bezier(.25,.8,.4,1)'},
        {transform:'translateY(0px) scale(1,1)',offset:.88,easing:'cubic-bezier(.3,0,.3,1)'},
        {transform:'translateY(0px) scale(1,1)',offset:1}],{duration:3000,delay:delay}];
    }
    var A=drop(-44,0),B=drop(-54,110);
    return {defs:[[og,A[0],A[1]],[ig,B[0],B[1]]],leap:true};
  }
  /* 10) 표창 — 날카롭게 가속해 두 바퀴를 돌고, 정위치에서 딱 멎는다(끝에 16° 반동 한 번). */
  function p9(){
    function spin(dir,delay){
      var a=720*dir,over=(720+18)*dir,back=(720-6)*dir;
      return [[
        {transform:'rotate(0deg) scale(1)',offset:0,easing:'cubic-bezier(.5,0,.85,.35)'},
        {transform:'rotate('+(over*.5)+'deg) scale(.965)',offset:.34,easing:'linear'},
        {transform:'rotate('+over+'deg) scale(1)',offset:.70,easing:'cubic-bezier(.3,.9,.4,1)'},
        {transform:'rotate('+back+'deg) scale(1)',offset:.82,easing:'cubic-bezier(.3,.9,.4,1)'},
        {transform:'rotate('+a+'deg) scale(1)',offset:1}],{duration:2200,delay:delay}];
    }
    var A=spin(1,0),B=spin(-1,80);
    return {defs:[[og,A[0],A[1]],[ig,B[0],B[1]]]};
  }
  var PAT=[p0,p1,p2,p3,p4,p5,p6,p7,p8,p9];
  /* 정지 상태로 되돌린다 — 애니메이션을 걷어 내고(모든 패턴이 끝 자리 = 시작 자리라 캔슬해도 튀지 않는다),
     인라인으로 만졌던 dash 값도 지워 정지 상태의 그림과 똑같이 맞춘다. */
  function settle(){
    clearTimeout(endT);endT=0;
    for(var i=0;i<runs.length;i++){try{runs[i].cancel();}catch(e){}}
    runs=[];busy=false;
    if(reset){try{reset();}catch(e){}reset=null;}
    lock.classList.remove('cctt-leap');
  }
  function play(i){
    if(reduce())return false;
    if(busy)return false;
    i=((i|0)%10+10)%10;
    var plan=PAT[i](),defs=plan.defs||plan,max=0,n=defs.length,k;
    busy=true;idx=i;minute=new Date().getMinutes();startedAt=performance.now();
    if(plan&&plan.leap)lock.classList.add('cctt-leap');
    if(plan&&plan.reset)reset=plan.reset;
    for(k=0;k<n;k++){runs.push(defs[k][0].animate(defs[k][1],defs[k][2]));}
    /* 끝난 뒤에는 애니메이션 객체를 스스로 버려 쉬는 동안 아무 일도 하지 않는다 */
    endT=setTimeout(function(){settle();},max+80);
    return true;
  }
  /* 방치 유도 감지 — 속성이 붙는 순간이 곧 심전도가 움직이기 시작하는 순간이다. */
  function watch(el){
    if(!el||!window.MutationObserver)return;
    new MutationObserver(function(m){
      for(var i=0;i<m.length;i++){
        var on=el.getAttribute('data-st')==='hint';
        if(on){if(!busy)play(new Date().getMinutes()%10);}
        else if(busy){settle();}      /* 도중에 손이 닿아 방치 유도가 접히면 마름모도 그 자리에서 멎는다 */
        return;
      }
    }).observe(el,{attributes:true,attributeFilter:['data-st']});
  }
  watch(document.getElementById('ecgRow'));
  watch(document.getElementById('slot'));
  window.JC_LOGO={play:play,stop:function(){settle();},names:function(){return NAMES.slice();},
    state:function(){return{busy:busy,pattern:idx,name:NAMES[idx]||'',minute:minute,startedAt:startedAt};}};
})();
</script>
